cmake_minimum_required(VERSION 3.21)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
project(Impasto VERSION 0.11.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Svg)
# Zip for OpenRaster and deflate for Photoshop channels. Windows builds may
# fetch a pinned zlib when no development package is installed.
find_package(ZLIB QUIET)
option(IMPASTO_FETCH_ZLIB "Download zlib when it is not installed" ${WIN32})
if(NOT ZLIB_FOUND)
    if(NOT IMPASTO_FETCH_ZLIB)
        find_package(ZLIB REQUIRED)
    endif()
    include(FetchContent)
    set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(impasto_zlib
        URL https://github.com/madler/zlib/archive/refs/tags/v1.3.1.tar.gz
        URL_HASH SHA256=17e88863f3600672ab49182f217281b6fc4d3c762bde361935e436a95214d05c)
    FetchContent_MakeAvailable(impasto_zlib)
    set_property(DIRECTORY "${impasto_zlib_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
    target_include_directories(zlibstatic PUBLIC
        "${impasto_zlib_SOURCE_DIR}" "${impasto_zlib_BINARY_DIR}")
    add_library(ZLIB::ZLIB ALIAS zlibstatic)
endif()

file(GLOB_RECURSE IMPASTO_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/src/*.cpp
    ${CMAKE_SOURCE_DIR}/src/*.h)

qt_add_executable(impasto
    ${IMPASTO_SOURCES}
    resources/resources.qrc)
# Host rcc may default to zstd; every supported Qt build provides zlib.
set_property(TARGET impasto PROPERTY AUTORCC_OPTIONS "--compress-algo;zlib")

target_include_directories(impasto PRIVATE src)
if(MSVC)
    target_compile_options(impasto PRIVATE /W4 /permissive- /utf-8 /bigobj)
else()
    target_compile_options(impasto PRIVATE -Wall -Wextra -Wno-unused-parameter)
    if(MINGW)
        target_compile_options(impasto PRIVATE -Wa,-mbig-obj)
    endif()
endif()
if(WIN32)
    enable_language(RC)
    set_target_properties(impasto PROPERTIES WIN32_EXECUTABLE TRUE)
    configure_file(packaging/windows/impasto.rc.in windows-impasto.rc @ONLY)
    target_sources(impasto PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows-impasto.rc")
    target_compile_definitions(impasto PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_compile_definitions(impasto PRIVATE
    IMPASTO_VERSION="${PROJECT_VERSION}")

target_link_libraries(impasto PRIVATE
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets Qt6::Svg ZLIB::ZLIB)

# Renders the app icon at the sizes desktops want (packaging/icons); built only when asked for.
add_executable(impasto-svg2png EXCLUDE_FROM_ALL tools/svg2png.cpp)
target_link_libraries(impasto-svg2png PRIVATE Qt6::Gui Qt6::Svg)

# ---------------------------------------------------------------- install
set(IMPASTO_APP_ID io.github.ozzy_apps.Impasto)
include(GNUInstallDirs)
if(WIN32)
    install(TARGETS impasto RUNTIME DESTINATION .)
    install(FILES LICENSE packaging/windows/README.txt DESTINATION .)
    install(DIRECTORY packaging/windows/licenses/ DESTINATION licenses)
    set(IMPASTO_WEBP_RUNTIME_DIR "" CACHE PATH "Optional external WebP runtime DLL directory")
    if(IMPASTO_WEBP_RUNTIME_DIR)
        install(DIRECTORY "${IMPASTO_WEBP_RUNTIME_DIR}/" DESTINATION .
            FILES_MATCHING PATTERN "*.dll")
    endif()
    option(IMPASTO_DEPLOY_QT "Bundle Qt and compiler runtime when installing" ON)
    if(IMPASTO_DEPLOY_QT)
        if(CMAKE_CROSSCOMPILING)
            message(FATAL_ERROR "Cross builds must set IMPASTO_DEPLOY_QT=OFF and stage the Windows Qt runtime separately")
        endif()
        get_filename_component(_qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
        set(_qt_bin "${_qt_prefix}/bin")
        find_program(IMPASTO_WINDEPLOYQT windeployqt HINTS "${_qt_bin}" REQUIRED)
        configure_file(packaging/windows/deploy.cmake.in windows-deploy.cmake @ONLY)
        install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/windows-deploy.cmake")
    endif()
else()
    install(TARGETS impasto RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES packaging/${IMPASTO_APP_ID}.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES packaging/${IMPASTO_APP_ID}.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
    install(FILES packaging/${IMPASTO_APP_ID}.mime.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages RENAME ${IMPASTO_APP_ID}.xml)
    install(FILES resources/icons/app.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME ${IMPASTO_APP_ID}.svg)
    foreach(size 16 24 32 48 64 128 256 512)
        install(FILES packaging/icons/${size}.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps RENAME ${IMPASTO_APP_ID}.png)
    endforeach()
endif()

# ---------------------------------------------------------------- packages
set(CPACK_PACKAGE_NAME Impasto)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_VENDOR "ozzy-apps")
set(CPACK_PACKAGE_CONTACT "ozzy-apps <info@ozzymail.nl>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Painting app for pen tablets")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/ozzy-apps/impasto")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The Windows package requires a 64-bit compiler")
    endif()
    set(CPACK_GENERATOR "ZIP;NSIS")
    set(CPACK_PACKAGE_FILE_NAME "Impasto-${PROJECT_VERSION}-windows-x64")
    set(CPACK_PACKAGE_INSTALL_DIRECTORY "Impasto")
    set(CPACK_PACKAGE_EXECUTABLES "impasto;Impasto")
    set(CPACK_CREATE_DESKTOP_LINKS "impasto")
    set(CPACK_NSIS_DISPLAY_NAME "Impasto")
    set(CPACK_NSIS_PACKAGE_NAME "Impasto")
    set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
    set(CPACK_NSIS_MUI_ICON "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/impasto.ico")
    set(CPACK_NSIS_MUI_UNIICON "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/impasto.ico")
    set(CPACK_NSIS_INSTALLED_ICON_NAME "impasto.exe")
    set(CPACK_NSIS_MUI_FINISHPAGE_RUN "impasto.exe")
    set(CPACK_NSIS_MANIFEST_DPI_AWARE ON)
    set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
    set(CPACK_NSIS_MODIFY_PATH OFF)
    set(CPACK_NSIS_BRANDING_TEXT " ")
    set(CPACK_NSIS_COMPRESSOR "/SOLID lzma")
    set(CPACK_NSIS_URL_INFO_ABOUT "${CPACK_PACKAGE_HOMEPAGE_URL}")
else()
    set(CPACK_GENERATOR DEB)
    set(CPACK_PACKAGE_NAME impasto)
    set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_DEBIAN_PACKAGE_SECTION graphics)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    # Qt loads these as plugins, so the library scan cannot see them.
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins")
    set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland, qt6-image-formats-plugins")
endif()
include(CPack)
